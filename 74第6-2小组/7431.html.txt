  <script>
    // 页面路由控制
    const pages = document.querySelectorAll('.page');
    const navButtons = document.querySelectorAll('.nav-card, .back-btn');

    function navigateTo(targetId) {
      pages.forEach(page => {
        page.classList.remove('active');
      });
      const target = document.getElementById(targetId);
      if (target) {
        target.classList.add('active');
        window.scrollTo({ top: 0, behavior: 'smooth' });
      }
    }

    navButtons.forEach(btn => {
      btn.addEventListener('click', (e) => {
        e.preventDefault();
        const target = btn.getAttribute('data-target');
        navigateTo(target);
      });
    });

    // 动物声音互动区 (Web Audio API 合成模拟)
    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const soundButtons = document.querySelectorAll('.sound-btn');

    const soundPresets = {
      cat: { freq: [600, 800], type: 'sine', dur: 0.15, pattern: 'up' },
      dog: { freq: [400, 300], type: 'triangle', dur: 0.2, pattern: 'down' },
      bird: { freq: [1200, 1500, 1200], type: 'square', dur: 0.08, pattern: 'chirp' },
      frog: { freq: [150, 200], type: 'sawtooth', dur: 0.25, pattern: 'croak' },
      owl: { freq: [350, 300, 350], type: 'sine', dur: 0.3, pattern: 'hoot' },
      whale: { freq: [80, 60], type: 'sine', dur: 0.8, pattern: 'long' }
    };

    function playAnimalSound(preset) {
      if (audioCtx.state === 'suspended') audioCtx.resume();
      
      const now = audioCtx.currentTime;
      let delay = 0;
      
      const frequencies = preset.freq;
      for (let i = 0; i < frequencies.length; i++) {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        
        osc.type = preset.type;
        osc.frequency.setValueAtTime(frequencies[i], now + delay);
        gain.gain.setValueAtTime(0.3, now + delay);
        gain.gain.exponentialRampToValueAtTime(0.01, now + delay + preset.dur);
        
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start(now + delay);
        osc.stop(now + delay + preset.dur + 0.1);
        
        delay += preset.dur + 0.05;
      }
    }

    soundButtons.forEach(btn => {
      btn.addEventListener('click', () => {
        const sound = btn.getAttribute('data-sound');
        const preset = soundPresets[sound];
        if (preset) {
          playAnimalSound(preset);
          btn.classList.add('playing');
          setTimeout(() => btn.classList.remove('playing'), 600);
        }
      });
    });

    // 宠物日记表单处理
    const petForm = document.getElementById('petForm');
    const petFeed = document.getElementById('petFeed');
    const placeholderImg = 'https://bailian-bmp-pre.oss-cn-hangzhou.aliyuncs.com/public/system_agent/PlaceHolder.png';

    // 初始化一些示例数据
    const samplePosts = [
      { name: '布丁', title: '第一次学会握手', content: '教了三周终于成功！奖励了半根猫条，它现在天天追着我要零食。', date: '2023-10-24' },
      { name: '奥利奥', title: '拆家记录', content: '出门一趟回来，沙发垫变成了雪花。但看到它无辜的眼神实在生不起气。', date: '2023-10-22' }
    ];

    function renderPost(post, prepend = false) {
      const div = document.createElement('div');
      div.className = 'pet-post';
      div.innerHTML = `
        <img src="${placeholderImg}" alt="宠物照片">
        <h4>${post.name} · ${post.title}</h4>
        <p>${post.content}</p>
        <div class="pet-date">发布于 ${post.date}</div>
      `;
      if (prepend) petFeed.insertBefore(div, petFeed.firstChild);
      else petFeed.appendChild(div);
    }

    samplePosts.forEach(post => renderPost(post));

    petForm.addEventListener('submit', (e) => {
      e.preventDefault();
      const inputs = petForm.querySelectorAll('input, textarea');
      const newPost = {
        name: inputs[0].value,
        title: inputs[1].value,
        content: inputs[2].value,
        date: new Date().toISOString().split('T')[0]
      };
      renderPost(newPost, true);
      petForm.reset();
      alert('日记发布成功！(当前为前端模拟展示)');
    });
  </script>
